A primary button is the filled one the game uses for Play: a gradient background with a border, for the one action
a screen is about. The variants mirror the plain buttons. The showcase's *Primary buttons* page:

<Playground preview="/assets/sdk/primary-buttons.webp">
```csharp excerpt
new BsPrimaryTextButton { Text = "Play", sColors = colors },
new BsPrimaryTextButton { Text = "Straight", Skew = 0f, sColors = colors },
new BsPrimaryTextButton {
    Text = "Maybe",
    sInteractable = interactable,
    sColors = colors
},

new BsPrimaryImageButton {
    FlexItem = {
        Size = 8.pt
    },

    Sprite = GameResources.SearchIcon,
    PreserveAspect = true,
    sColors = colors
}
```
```csharp
class PrimaryButtons : ReactiveComponent {
    protected override GameObject Construct() {
        var interactable = Remember(false);

        return new Layout {
            FlexController = {
                AlignItems = Align.Center,
                JustifyContent = Justify.Center,
                Gap = 2.pt
            },

            Children = {
                new BsPrimaryTextButton { Text = "Play" },
                new BsPrimaryTextButton { Text = "Straight", Skew = 0f },
                new BsPrimaryTextButton {
                    Text = "Maybe",
                    sInteractable = interactable
                },

                new BsPrimaryImageButton {
                    FlexItem = {
                        Size = 8.pt
                    },

                    Sprite = GameResources.SearchIcon,
                    PreserveAspect = true
                }
            }
        }.Use();
    }
}
```

`BsPrimaryButtonColors` has gradient colors, border colors and outline colors instead of the plain button's
background and underline, and otherwise the API is the same. A non-interactable primary button falls back to the
`NotInteractable` color of each of those sets, so the greyed-out one still looks like a button.

The showcase's example that's worth copying is a button that becomes a spinner while something loads:

<Playground preview="/assets/sdk/primary-loading.webp">
```csharp excerpt
var phase = Remember(Phase.Idle);

new BsPrimaryButton {
    FlexItem = {
        Size = new() { x = 36.pt, y = 8.pt }
    },

    FlexController = {
        JustifyContent = Justify.Center,
        AlignItems = Align.Center,
        Gap = 1.pt
    },

    sColors = colors,
    sInteractable = phase.Map(x => x == Phase.Idle),
    OnClick = () => StartCoroutine(RunFakeTask()),

    ConstructContent = (color, _) => new Layout {
        FlexController = {
            AlignItems = Align.Center,
            Gap = 1.pt
        },

        Children = {
            new Spinner {
                FlexItem = {
                    Size = 4.pt
                },

                sEnabled = phase.Map(x => x == Phase.Loading),
                Image = {
                    sColor = color.In()
                }
            },

            new Label {
                sText = phase.Map(x => x switch {
                    Phase.Loading => "Loading",
                    Phase.Done => "Done",
                    _ => "Run"
                }),
                sColor = color.In()
            }
        }
    }
}
```
```csharp
class LoadingButton : ReactiveComponent {
    private enum Phase { Idle, Loading, Done }

    private State<Phase> _phase = null!;

    private IEnumerator RunFakeTask() {
        _phase.Value = Phase.Loading;
        yield return new WaitForSeconds(1.4f);
        _phase.Value = Phase.Done;
        yield return new WaitForSeconds(1f);
        _phase.Value = Phase.Idle;
    }

    protected override GameObject Construct() {
        var phase = _phase = Remember(Phase.Idle);

        return new Layout {
            FlexController = {
                AlignItems = Align.Center,
                JustifyContent = Justify.Center
            },

            Children = {
                new BsPrimaryButton {
                    FlexItem = {
                        Size = new() { x = 36.pt, y = 8.pt }
                    },

                    FlexController = {
                        JustifyContent = Justify.Center,
                        AlignItems = Align.Center,
                        Gap = 1.pt
                    },

                    sInteractable = phase.Map(x => x == Phase.Idle),
                    OnClick = () => StartCoroutine(RunFakeTask()),

                    ConstructContent = (color, _) => new Layout {
                        FlexController = {
                            AlignItems = Align.Center,
                            Gap = 1.pt
                        },

                        Children = {
                            new Spinner {
                                FlexItem = {
                                    Size = 4.pt
                                },

                                sEnabled = phase.Map(x => x == Phase.Loading),
                                Image = {
                                    sColor = color.In()
                                }
                            },

                            new Label {
                                sText = phase.Map(x => x switch {
                                    Phase.Loading => "Loading",
                                    Phase.Done => "Done",
                                    _ => "Run"
                                }),
                                sColor = color.In()
                            }
                        }
                    }
                }
            }
        }.Use();
    }
}
```
</Playground>


One state, `phase`, drives the interactable flag, the spinner's visibility and the label's text, and the coroutine
only writes `phase`. The button is given a fixed size so it doesn't resize when the spinner appears. `Spinner`
exposes its `Image` for a nested initializer, which is how it picks up the button's content color.
